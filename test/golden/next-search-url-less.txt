import type { Metadata } from 'next';

import { Providers } from '../components/providers/AppProviders';
import { SITE } from '../lib/site.config';
import '../styles/globals.css';

/**
 * The root layout: the top of the tree, and the only place `<html>` exists.
 *
 * A server component, deliberately, and it stays one. Nothing here needs state,
 * effects or browser APIs, so nothing here needs to reach the browser as
 * JavaScript - and marking this file `'use client'` would pull every page under
 * it into the client bundle to solve a problem it does not have.
 *
 * ## The provider boundary
 *
 * `Providers` is always here, whether or not anything fills it. A project
 * with no UI library gets a pass-through that renders its children and nothing
 * else; a project with one gets that library's providers instead, at the same
 * place, without this file changing. Where the client boundary goes - if one is
 * needed at all - is that component's decision, which is what keeps it out of
 * this one.
 *
 * `metadata` starts as a title and a description taken from the one config
 * file. A canonical address is declared here only when the project was
 * generated with `--features seo`, and only once `SITE.url` is set. There is no
 * Open Graph block and no social card: those need a considered decision per
 * page, and inventing one would put claims in the head that nobody made.
 */
export const metadata: Metadata = {
  title: SITE.name,
  description: SITE.description,
};

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang={SITE.locale || 'en'}>
      <body>
        <Providers>{children}</Providers>
      </body>
    </html>
  );
}

import type { MetadataRoute } from 'next';

import { SITE } from '../lib/site.config';
import { absoluteUrl, siteOrigin } from '../lib/seo';

/**
 * `/robots.txt`, served by Next from this route module.
 *
 * Every crawler may index every page. The sitemap is advertised only once
 * `SITE.url` is a real address, because a `Sitemap:` line has to be absolute
 * and no domain is ever guessed. A missing URL does not block indexing either:
 * a site without a domain yet is not a site that asked to be hidden.
 */
export default function robots(): MetadataRoute.Robots {
  const origin = siteOrigin(SITE.url);

  return {
    rules: { userAgent: '*', allow: '/' },
    ...(origin === '' ? {} : { sitemap: absoluteUrl(origin, '/sitemap.xml') }),
  };
}

import type { MetadataRoute } from 'next';

import { SITE } from '../lib/site.config';
import { absoluteUrl, siteOrigin } from '../lib/seo';

/**
 * Every page this sitemap lists, as site-relative paths.
 *
 * Next does not discover routes for a sitemap, so this list is the whole
 * truth: add a path here when you add a page that should be crawled. The
 * not-found page is deliberately absent - it is a status, not content.
 */
const ROUTES: readonly string[] = ['/'];

/**
 * `/sitemap.xml`, served by Next from this route module.
 *
 * Empty until `SITE.url` is a real address: a sitemap entry has to be an
 * absolute URL, and no domain is ever guessed. No `lastModified` either - a
 * date read from the clock at build time would claim every page changed on
 * every deploy.
 */
export default function sitemap(): MetadataRoute.Sitemap {
  const origin = siteOrigin(SITE.url);
  if (origin === '') return [];

  return ROUTES.map((route) => ({ url: absoluteUrl(origin, route) }));
}

import { CONTACT, SITE } from '../lib/site.config';
import { jsonLdScript, siteOrigin } from '../lib/seo';

/**
 * schema.org Organization data, built only from values configured in
 * `lib/site.config.ts`.
 *
 * Search engines consume structured data automatically, so a fabricated field
 * is worse than a missing one: every property below is omitted unless it has
 * a real value. No logo, address, social profile, rating or founding date is
 * invented.
 *
 * A server component, rendered into the page's markup so it is in the
 * response a crawler reads. The home page renders it; the not-found page
 * deliberately does not. Render `<StructuredData />` on any page you add that
 * should describe the organisation.
 */
export function StructuredData() {
  const origin = siteOrigin(SITE.url);
  const description = SITE.description.trim();
  const email = CONTACT.email.trim();
  const phone = CONTACT.phone.trim();

  const organization: Record<string, unknown> = {
    '@context': 'https://schema.org',
    '@type': 'Organization',
    name: SITE.name,
  };

  if (origin !== '') organization['url'] = origin;
  if (description !== '') organization['description'] = description;
  if (email !== '') organization['email'] = email;
  if (phone !== '') organization['telephone'] = phone;

  return (
    <script
      type="application/ld+json"
      // Escaped for a script element: see jsonLdScript in lib/seo.ts.
      dangerouslySetInnerHTML={{ __html: jsonLdScript(organization) }}
    />
  );
}

import { SITE } from '../../lib/site.config';

/**
 * A placeholder brand mark: the client's initials in a rounded square.
 *
 * Deliberately not an image. A generated project has no logo file, and shipping
 * a stand-in graphic would be something to find and delete later; initials read
 * as intentional until a real mark replaces them.
 *
 * The class names are the project's shared style contract - `brand`,
 * `brand-lg`, `brand-mark`, `brand-name` - which every styling system
 * implements. That is what lets this component render under plain CSS or
 * Tailwind without knowing which was selected.
 */
export function Mark({ size = 'lg' }: { size?: 'sm' | 'lg' }) {
  const initials = SITE.name
    .split(/\s+/)
    .filter(Boolean)
    .slice(0, 2)
    .map((word) => word[0])
    .join('')
    .toUpperCase();

  return (
    <span className={`brand brand-${size}`}>
      <span className="brand-mark" aria-hidden="true">
        {initials || '—'}
      </span>
      <span className="brand-name">{SITE.name}</span>
    </span>
  );
}

/**
 * The URL rules every search file in this project shares.
 *
 * `app/robots.ts`, `app/sitemap.ts` and the structured data all build absolute
 * addresses from `SITE.url`, and they have to agree: a sitemap listing one
 * domain while robots.txt advertises another is worse than either alone. So
 * the arithmetic lives here once.
 */

/**
 * The site's origin - scheme, host and port - or '' when no usable production
 * URL is configured.
 *
 * Origin only, on purpose: a path, query or fragment in `SITE.url` is dropped
 * rather than repeated in front of every address. Anything that is not an
 * absolute http(s) URL counts as not configured, so nothing absolute is
 * emitted until there is a real domain to point at.
 */
export function siteOrigin(url: string): string {
  const trimmed = url.trim();
  if (trimmed === '') return '';

  let parsed: URL;
  try {
    parsed = new URL(trimmed);
  } catch {
    return '';
  }
  if (parsed.protocol !== 'http:' && parsed.protocol !== 'https:') return '';

  return parsed.origin;
}

/**
 * Joins a site-relative path onto an origin, or '' when there is no origin -
 * so a caller can omit the address entirely rather than invent one.
 */
export function absoluteUrl(origin: string, path: string): string {
  if (origin === '') return '';
  const suffix = path.startsWith('/') ? path : `/${path}`;
  return `${origin}${suffix}`.replace(/([^:]\/)\/+/g, '$1');
}

/**
 * The three characters that can end a script element early or open a new tag,
 * mapped to their JSON unicode escapes.
 *
 * Each value is six characters starting with a real backslash. Writing the
 * character itself would make the replacement below a silent no-op.
 */
const JSON_LD_ESCAPES: Readonly<Record<string, string>> = {
  '<': '\\u003c',
  '>': '\\u003e',
  '&': '\\u0026',
};

/**
 * Serialises data for a `<script type="application/ld+json">` element.
 *
 * `JSON.stringify` alone is not enough: the HTML parser ends a script element
 * at the first `</script`, whatever the JSON quoting says. Escaping `<`, `>`
 * and `&` keeps the payload valid JSON - the escapes decode back to the
 * original characters - while making it impossible to close the element.
 */
export function jsonLdScript(data: unknown): string {
  return JSON.stringify(data).replace(/[<>&]/g, (char) => JSON_LD_ESCAPES[char] ?? char);
}

# Acme Ltd

Bespoke widgets.

Built with [Next.js](https://nextjs.org) (App Router), TypeScript and plain CSS.

## Getting started

```bash
npm install
npm run dev
```

The site runs at http://localhost:3000.

## Scripts

| Script              | What it does                          |
| ------------------- | ------------------------------------- |
| `npm run dev`       | Development server with hot reloading |
| `npm run build`     | Production build                      |
| `npm run start`     | Serves the production build           |
| `npm run typecheck` | Type-checks without emitting          |

There is no `lint` script. Next 16 removed the `next lint` command, and adding
one would mean adding ESLint and its Next configuration - a choice better made
by you than assumed here.

## Where things live

```
app/
  layout.tsx      the root layout: <html>, <body>, baseline metadata
  page.tsx        the home page
components/ui/    presentational pieces
lib/
  site.config.ts  everything client-specific, in one file
public/           static assets served from /
styles/
  globals.css     the only stylesheet; custom properties are the design system
```

## First things to change

- `lib/site.config.ts` — name, description, locale and contact all live there.
- Set `SITE.url` once the domain is known. It is empty on purpose: nothing
  absolute is emitted until it is filled in, so no page points at a domain
  nobody owns.
- Replace `public/favicon.svg` with the client mark.

## What is deliberately not here

No CSS framework, no component library, no client-side router, no analytics and
no deployment configuration. Each of those is a decision with consequences, and
a scaffold that makes them for you is one you have to undo first.
